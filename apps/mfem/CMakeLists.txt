find_package(MFEM 4.8 CONFIG REQUIRED)
if(NOT MFEM_USE_CUDA OR NOT MFEM_USE_DOUBLE OR MFEM_USE_MPI)
    message(FATAL_ERROR "mfem_heat requires a serial FP64 CUDA build of MFEM")
endif()
add_executable(mfem_heat heat.cpp sparseSum.cu)
target_include_directories(mfem_heat PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mfem_heat PRIVATE mfem atspmv_plan)
target_compile_features(mfem_heat PRIVATE cxx_std_17 cuda_std_17)
add_executable(mfem_sparse_sum_test EXCLUDE_FROM_ALL ../../tests/mfem_sparse_sum_test.cpp sparseSum.cu)
target_include_directories(mfem_sparse_sum_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mfem_sparse_sum_test PRIVATE mfem CUDA::cudart)
target_compile_features(mfem_sparse_sum_test PRIVATE cxx_std_17 cuda_std_17)
set_target_properties(mfem_heat mfem_sparse_sum_test PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
add_executable(packed_spmv_bench EXCLUDE_FROM_ALL ../../bench/packed_spmv_bench.cu)
target_link_libraries(packed_spmv_bench PRIVATE atspmv_plan)
set_target_properties(packed_spmv_bench PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
